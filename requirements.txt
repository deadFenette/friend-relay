# Friend Relay v5 — зависимости
#
# Установка: pip install -r requirements.txt
# (на Windows: py -m pip install -r requirements.txt)
#
# Все пакеты ниже используются приложением. cryptography и sounddevice
# опциональны — приложение работает и без них (с урезанной функциональностью):
#   - без cryptography: шифрование сообщений падает обратно на XOR (небезопасно,
#     но не падает)
#   - без sounddevice: голосовой канал недоступен (UI показывает ошибку)

# GUI фреймворк (обязательно)
# (v3.5.0) все версии с ВЕРХНЕЙ границей: мажорный релиз (например
# PySide6 7.0) может молча сломать API — обновлям мажоры руками и
# осознанно, минорные/патч-обновления pip подтянет сам.
PySide6>=6.5.0,<7

# HTTP keep-alive пул соединений (опционально, лицензия MIT).
# Без него клиент работает через urllib без переиспользования соединений:
# медленнее поллинг и переключение каналов (см. lib/client.py, v9.3).
urllib3>=2.0.0,<3

# Быстрый JSON для сервера (опционально, лицензия MIT, v2.0.2).
# Без него всё работает на stdlib json (см. lib/json_fast.py), просто
# медленнее: orjson в 5-10 раз быстрее парсит/сериализует — полезнее всего
# хосту с большим журналом (каждый /events poll и запись в history.jsonl).
# Установка: pip install orjson
orjson>=3.9.0,<4

# Криптография для AES-256-GCM + PBKDF2 (опционально, fallback на XOR)
cryptography>=41.0.0,<48

# Захват/воспроизведение звука для голосового канала (опционально)
sounddevice>=0.4.6,<1

# Массивы для аудио-миксера (опционально, но рекомендуется
# для голосового канала — без него миксер работает медленно)
numpy>=1.24.0,<3

# Голос в браузере: WSS-мост web-клиент <-> VoiceMixer (опционально —
# без него чат/файлы/шахматы работают, а голос в браузере недоступен)
websockets>=12.0,<17

# Opus-кодек голоса (опционально, v2.0.0): fullband 48 кГц, Opus VOIP
# с FEC/PLC на 48 кбит/с вместо raw PCM 768 кбит/с. Лицензии:
# opuslib_next — BSD-3-Clause, libopus — BSD-3-Clause. БЕЗ пакета
# (или без libopus) голос автоматически работает как раньше — raw PCM,
# ничего не падает.
# Windows: кроме pip, положи libopus-0.dll (x64) в папку приложения
# (рядом с RUN.bat) или в PATH — voice/codec.py найдёт её сам.
opuslib_next>=1.3.0,<2

# Метаданные музыки: длительность/теги треков для синхронного плеера
# (вкладка «Музыка», v3.3.0). ОПЦИОНАЛЬНО: без него музыка работает,
# а переход трека — по отчётам браузеров «докатился до конца».
# Установка: pip install mutagen
mutagen>=1.47.0,<2
